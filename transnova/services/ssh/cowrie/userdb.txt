# A leading "!" on the password denies; "*" matches anything.
# All accounts and passwords are synthetic lab values.
#

# produces cowrie.login.failed events.

# The weak "backup" account is the intended way in. It exists in the emulated
# /etc/passwd with home /var/backups, where the decoy files live.
